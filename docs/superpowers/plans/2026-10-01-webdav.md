```kotlin
    /**
     * Legal names, for the API and for WebDAV alike.
     *
     * "/" is rejected because a name is one path segment - both protocols
     * address a resource by joining encoded segments with slashes.
     *
     * "." and ".." are rejected because they are relative-path directives, not
     * names: a filesystem-shaped client resolves "/a/../b" as "/b", so a real
     * folder called ".." makes the intent ambiguous, and textually resolving a
     * path that contains one is how a request walks out of the user's root.
     *
     * C0 controls and DEL are rejected because XML 1.0 cannot carry them raw.
     * A single such name turns a WebDAV Multi-Status listing into a document
     * no client can parse, which hides the entire directory rather than the one
     * broken file. Rejecting at the door also keeps them out of logs and
     * terminal output everywhere else.
     */
    private fun nameOk(name: String) =
        name.isNotBlank() &&
            name.length <= 255 &&
            !name.contains('/') &&
            name != "." &&
            name != ".." &&
            name.none { it.code < 0x20 || it.code == 0x7F }
```

**关键设计（不要改）:**

- 令牌是**高熵随机串**（32 字节 → 64 位十六进制），**按 SHA-256 精确查，不做 bcrypt**。这正是它比账号密码快的唯一原因：账号密码是 bcrypt cost=10 ≈ 100ms/次，而挂载后每个请求都要付一次。令牌不可爆破，所以精确匹配没有离线爆破面。
- `/dav` 的失败计数与登录额度**彻底分开**：挂载点的每个请求都带凭据，客户端重试循环若走登录额度会烧光 10 次/分钟并触发 20 次/15min 锁定，把挂载自己锁死（RFC 7617 附录明确警告过这个）。
- 令牌**不能**登录 Web UI、不能调管理端点（只认 Bearer JWT）。

```kotlin
            contentType(ContentType.Application.Json); setBody("""{"username":"user1","password":"password123"}""")

            client.post("/api/auth/login") {
                contentType(ContentType.Application.Json); setBody("""{"username":"user1","password":"password123"}""")
            }.bodyAsText()
        )!!.groupValues[1]
    }

    private suspend fun ApplicationTestBuilder.createToken(label: String = "iPad", readOnly: Boolean = false) =
```

```kotlin
/**
 * One revocable credential per mounted device.
 *
 * Not a password: the value is 32 bytes of randomness, stored as a SHA-256
 * hash and looked up by exact match. That is what makes it usable for WebDAV -
 * a mount authenticates on every request, and the account password is a bcrypt
 * hash costing ~100ms, so browsing one folder would spend minutes in KDF.
 * A high-entropy token has no offline guessing surface, so exact-match storage
 * is the right trade here and bcrypt would only add latency.
 */
object WebdavTokensTable : Table("webdav_tokens") {
    val id = uuid("id")
    val user = uuid("user_id").references(UsersTable.id)
    /** SHA-256 hex. Exact match on lookup: see the type note above. */
    val tokenHash = varchar("token_hash", 64)
    /** What the user calls this device, so a stray token is recognisable. */
    val label = varchar("label", 64)
    /** A read-only mount cannot delete the library through a stray keystroke. */
    val readOnly = bool("read_only").default(false)
    val lastUsedAt = long("last_used_at").clientDefault { 0L }
    val createdAt = long("created_at").clientDefault { System.currentTimeMillis() }
    override val primaryKey = PrimaryKey(id)
    init {
        uniqueIndex(tokenHash)
        index(customIndexName = "webdav_tokens_user_idx", isUnique = false, user)
    }
}
```

```kotlin
package com.linan.barezen_drive.webdav

import com.linan.barezen_drive.db.DatabaseFactory
import com.linan.barezen_drive.db.WebdavTokensTable
import com.linan.barezen_drive.files.FileService
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import org.jetbrains.exposed.sql.SqlExpressionBuilder.eq
import org.jetbrains.exposed.sql.deleteWhere
import org.jetbrains.exposed.sql.insert
import org.jetbrains.exposed.sql.selectAll
import org.jetbrains.exposed.sql.transactions.transaction
import java.security.MessageDigest
import java.security.SecureRandom
import java.util.UUID

/** A credential that authenticated a WebDAV request, as the routes need it. */
data class WebdavTokenRecord(val userId: UUID, val readOnly: Boolean, val id: UUID)

/** What the settings list shows. Carries no secret: the token is shown once. */
data class WebdavTokenView(val id: String, val label: String, val readOnly: Boolean, val createdAt: Long, val lastUsedAt: Long)

/**
 * App passwords for mounted drives.
 *
 * Deliberately not the account password. See [WebdavTokensTable] for why the
 * lookup is an exact hash match rather than a KDF: this credential exists
 * precisely so that per-request verification is cheap.
 */
object WebdavTokenService {
    private val random = SecureRandom()
    private val hex = "0123456789abcdef".toCharArray()

    private fun newToken(): String {
        val bytes = ByteArray(32)
        random.nextBytes(bytes)
        return buildString(64) {
            for (b in bytes) { append(hex[(b.toInt() shr 4) and 0xF]); append(hex[b.toInt() and 0xF]) }
        }
    }

    private fun hash(token: String): String =
        MessageDigest.getInstance("SHA-256").digest(token.toByteArray())
            .joinToString("") { (it.toInt() and 0xff).toString(16).padStart(2, '0') }

    suspend fun create(userId: UUID, label: String, readOnly: Boolean): Pair<WebdavTokenView, String> =
        withContext(Dispatchers.IO) {
            val token = newToken()
            val id = UUID.randomUUID()
            val now = System.currentTimeMillis()
            val dto = transaction(DatabaseFactory.db) {
                WebdavTokensTable.insert {
                    it[WebdavTokensTable.id] = id
                    it[WebdavTokensTable.user] = userId
                    it[WebdavTokensTable.tokenHash] = hash(token)
                    it[WebdavTokensTable.label] = label
                    it[WebdavTokensTable.readOnly] = readOnly
                    it[createdAt] = now
                }
                WebdavTokenView(id.toString(), label, readOnly, now, 0L)
            }
            dto to token
        }

    suspend fun list(userId: UUID): List<WebdavTokenView> = withContext(Dispatchers.IO) {
        transaction(DatabaseFactory.db) {
            WebdavTokensTable.selectAll().where { WebdavTokensTable.user eq userId }
                .map { row ->
                    WebdavTokenView(
                        row[WebdavTokensTable.id].toString(),
                        row[WebdavTokensTable.label],
                        row[WebdavTokensTable.readOnly],
                        row[WebdavTokensTable.createdAt],
                        row[WebdavTokensTable.lastUsedAt],
                    )
                }
        }
    }

    /** True when a row was actually removed, so 404 only for a token that is not yours. */
    suspend fun revoke(userId: UUID, tokenId: UUID): Boolean = withContext(Dispatchers.IO) {
        transaction(DatabaseFactory.db) {
            WebdavTokensTable.deleteWhere {
                (WebdavTokensTable.id eq tokenId) and (WebdavTokensTable.user eq userId)
            } > 0
        }
    }

    /**
     * Exact-match lookup, then stamp last_used_at so the settings list can show
     * which devices are still in use. The stamp is best effort: a failure to
     * record usage must not fail an otherwise valid request.
     */
    suspend fun authenticate(username: String, password: String): WebdavTokenRecord? =
        withContext(Dispatchers.IO) {
            val wanted = hash(password)
            transaction(DatabaseFactory.db) {
                val row = WebdavTokensTable.selectAll()
                    .where { WebdavTokensTable.tokenHash eq wanted }
                    .firstOrNull() ?: return@transaction null
                val userId = row[WebdavTokensTable.user]
                val name = transaction(DatabaseFactory.db) {
                    com.linan.barezen_drive.db.UsersTable.selectAll()
                        .where { com.linan.barezen_drive.db.UsersTable.id eq userId }
                        .single()[com.linan.barezen_drive.db.UsersTable.username]
                }
                if (name != username) return@transaction null
                runCatching {
                    WebdavTokensTable.update({ WebdavTokensTable.id eq row[WebdavTokensTable.id] }) {
                        it[lastUsedAt] = System.currentTimeMillis()
                    }
                }
                WebdavTokenRecord(userId, row[WebdavTokensTable.readOnly], row[WebdavTokensTable.id])
            }
        }
}
```

```kotlin
 * Token management for mounted drives. Ordinary Bearer endpoints under /api:
 * the browser session is a different credential from a WebDAV mount, and a headless

fun Route.webdavTokenRoutes() {

        route("/api/webdav/tokens") {

                call.respond(HttpStatusCode.NoContent)

        }
    }
```

```kotlin
package com.linan.barezen_drive.webdav

import com.linan.barezen_drive.db.DatabaseFactory
import com.linan.barezen_drive.db.FilesTable
import com.linan.barezen_drive.db.FoldersTable
import org.jetbrains.exposed.sql.SqlExpressionBuilder.eq
import org.jetbrains.exposed.sql.and
import org.jetbrains.exposed.sql.selectAll
import org.jetbrains.exposed.sql.transactions.transaction
import java.util.UUID

/**
 * One row of a WebDAV collection listing.
 *
 * Deliberately not a DTO: a 100k-file directory must never materialise 100k
 * of anything, so this carries only what a multistatus entry needs and the
 * listing walks it in keyset pages.
 */
data class DavEntry(
    val name: String,
    val collection: Boolean,
    val size: Long,
    val updatedAt: Long,
    val createdAt: Long,
    /** Null for collections: they have no content, so there is nothing to tag. */
    val etag: String?,
)

/**
 * Keyset paging over one collection, for PROPFIND.
 *
 * [FileService.contents] cannot serve this: it has no cursor for folders at
 * all, and with a null limit it maps the whole ordered file set to DTOs. A mount
 * always needs the complete listing (the client caches it), so the paging has
 * to live here where it can stay light.
 *
 * Ordered by (lowerCase(name), id) exactly like the album/contents pages, so
 * one page boundary can never straddle the sort.
 */
object DavListing {
    const val PAGE = 500

    private fun after(name: String?, id: UUID?): org.jetbrains.exposed.sql.Op<Boolean>? {
        if (name == null || id == null) return null
        val lower = FilesTable.name.lowerCase()
        return (lower greater name) or ((lower eq name) and (FilesTable.id greater id))
    }

    /** Folders first, then files, both name-ordered; the cursor spans the pair. */
    fun page(userId: UUID, parent: UUID?, afterName: String?, afterId: UUID?, limit: Int = PAGE): List<DavEntry> =
        transaction(DatabaseFactory.db) {
            val folders = FoldersTable.selectAll().where {
                (FoldersTable.user eq userId) and scoped(FoldersTable.parent, parent)
            }.map {
                DavEntry(
                    it[FoldersTable.name], true, 0L,
                    it[FoldersTable.updatedAt], it[FoldersTable.createdAt], null,
                )
            }
            var q = FilesTable.selectAll().where {
                (FilesTable.user eq userId) and scoped(FilesTable.folder, parent) and (FilesTable.deletedAt eq 0L)
            }
            after(afterName, afterId)?.let { q = q.andWhere { it } }
            val files = q.orderBy(FilesTable.name.lowerCase() to org.jetbrains.exposed.sql.SortOrder.ASC, FilesTable.id to org.jetbrains.exposed.sql.SortOrder.ASC)
                .limit(limit)
                .map {
                    DavEntry(
                        it[FilesTable.name], false, it[FilesTable.size],
                        it[FilesTable.updatedAt], it[FilesTable.createdAt],
                        // The blob key's last segment: content-addressed, so a
                        // strong ETag for free. The whole key would leak the
                        // storage layout (blobs/ab/cd/<sha>).
                        it[FilesTable.storageKey].substringAfterLast('/'),
                    )
                }
            (folders + files)
                .sortedWith(compareBy({ it.name.lowercase() }, { it.name }))
                .dropWhile { if (afterName == null) false else (it.name.lowercase() < afterName || (it.name.lowercase() == afterName && false)) }
                .take(limit)
        }

    private fun scoped(column: org.jetbrains.exposed.sql.Column<UUID?>, parent: UUID?): org.jetbrains.exposed.sql.Op<Boolean> =
        if (parent == null) column.isNull() else (column eq parent)
}
```

```kotlin
package com.linan.barezen_drive.webdav

import java.io.Writer
import java.time.Instant
import java.time.ZoneOffset
import java.time.format.DateTimeFormatter

/**
 * The two different encodings a name goes through, in one place so they cannot
 * drift:
 *
 *  - [hrefSegment] percent-encodes for the URI. A client resolves the path from
 *    it, so "#" would truncate the href and "?" would start a query string.
 *  - [xmlEscaped] only escapes the five XML metacharacters. displayname is
 *    human-readable text; percent-encoding it would show the user "a%20b.txt"
 *    in their file browser.
 */
object DavProperties {
    private const val UNRESERVED = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789-_.~"

    fun hrefSegment(name: String): String = buildString(name.length * 2) {
        for (b in name.toByteArray(Charsets.UTF_8)) {
            val ch = b.toInt().toChar()
            if (b >= 0 && ch in UNRESERVED) append(ch) else append('%').append("%02X".format(b))
        }
    }

    /** Escapes text content AND drops the characters XML 1.0 cannot represent. */
    fun xmlEscaped(s: String): String = buildString(s.length) {
        for (c in s) when {
            c == '&' -> append("&amp;")
            c == '<' -> append("&lt;")
            c == '>' -> append("&gt;")
            c == '"' -> append("&quot;")
            c == '\'' -> append("&apos;")
            // Task 1 rejects these at the door; the filter is the second line of
            // defence so a row written before that rule cannot emit invalid XML.
            c.code < 0x20 || c.code == 0x7F -> Unit
            else -> append(c)
        }
    }

    private val RFC1123: DateTimeFormatter =
        DateTimeFormatter.ofPattern("EEE, dd MMM yyyy HH:mm:ss 'GMT'", java.util.Locale.US)
            .withZone(ZoneOffset.UTC)

    fun httpDate(epochMillis: Long): String = RFC1123.format(Instant.ofEpochMilli(epochMillis))

    /**
     * One <D:response> entry. Written straight to [out] so a large listing never
     * has to exist as one string.
     */
    fun writeResponse(
        out: Writer,
        href: String,
        entry: DavEntry,
        requested: Set<String>,
    ) {
        out.write("<D:response><D:href>")
        out.write(xmlEscaped(href))
        out.write("</D:href><D:propstat><D:prop>")
        out.write("<D:resourcetype>")
        if (entry.collection) out.write("<D:collection/>")
        out.write("</D:resourcetype>")
        if ("" in requested || "displayname" in requested) {
            out.write("<D:displayname>"); out.write(xmlEscaped(entry.name)); out.write("</D:displayname>")
        }
        if ("" in requested || "getcontentlength" in requested) {
            out.write("<D:getcontentlength>"); out.write(if (entry.collection) "0" else entry.size.toString()); out.write("</D:getcontentlength>")
        }
        // updated_at, never taken_at: clients use this to decide what to refetch,
        // and a photo taken in 2021 would look like it had never been modified.
        if ("" in requested || "getlastmodified" in requested) {
            out.write("<D:getlastmodified>"); out.write(httpDate(entry.updatedAt)); out.write("</D:getlastmodified>")
        }
        if ("" in requested || "creationdate" in requested) {
            out.write("<D:creationdate>"); out.write(iso(entry.createdAt)); out.write("</D:creationdate>")
        }
        // Collections have no content, so there is nothing to tag.
        if (entry.etag != null && ("" in requested || "getetag" in requested)) {
            out.write("<D:getetag>\""); out.write(xmlEscaped(entry.etag)); out.write("\"</D:getetag>")
        }
        out.write("</D:prop><D:status>HTTP/1.1 200 OK</D:status></D:propstat>")
        // RFC 4918: a property that does not exist must be reported, or the client
        // cannot tell "no value" from "not looked at".
        val missing = requested - supported(entry) - "" - "displayname" - "getcontentlength" -
            "getlastmodified" - "creationdate" - "getetag"
        for (name in missing) {
            out.write("<D:propstat><D:prop><D:"); out.write(name)
            out.write("/></D:prop><D:status>HTTP/1.1 404 Not Found</D:status></D:propstat>")
        }
        out.write("</D:response>")
    }

    private fun supported(entry: DavEntry) = buildSet {
        add("resourcetype")
        add("getetag")
        if (!entry.collection) add("getcontentlength")
    }

    private fun iso(epochMillis: Long): String =
        DateTimeFormatter.ISO_INSTANT.format(Instant.ofEpochMilli(epochMillis))
}
```

```kotlin
package com.linan.barezen_drive.webdav

import com.linan.barezen_drive.db.DatabaseFactory
import com.linan.barezen_drive.db.FilesTable
import com.linan.barezen_drive.db.FoldersTable
import org.jetbrains.exposed.sql.SqlExpressionBuilder.eq
import org.jetbrains.exposed.sql.selectAll
import org.jetbrains.exposed.sql.transactions.transaction
import java.util.UUID

/** What a WebDAV path points at. */
sealed interface DavTarget {
    data class Collection(val id: UUID?) : DavTarget       // null id = the account root
    data class Blob(val fileId: UUID) : DavTarget
    data object Missing : DavTarget
}

/**
 * Resolve a path by walking one folder row per segment.
 *
 * Deliberately not a prefix match over the whole table: a per-segment walk is
 * `depth` indexed lookups, and it is the only way a name like "a/b" can never be
 * confused with a folder "a" containing a folder "b".
 */
fun resolveDavPath(userId: UUID, path: String): DavTarget {
    val segments = path.trim('/').split('/').filter { it.isNotEmpty() }
    if (segments.isEmpty()) return DavTarget.Collection(null)
    return transaction(DatabaseFactory.db) {
        var parent: UUID? = null
        for (i in 0 until segments.size - 1) {
            val next = FoldersTable.selectAll().where {
                (FoldersTable.user eq userId) and (FoldersTable.name eq segments[i]) and
                    if (parent == null) FoldersTable.parent.isNull() else (FoldersTable.parent eq parent)
            }.firstOrNull() ?: return@transaction DavTarget.Missing
            parent = next[FoldersTable.id]
        }
        val last = segments.last()
        FoldersTable.selectAll().where {
            (FoldersTable.user eq userId) and (FoldersTable.name eq last) and
                if (parent == null) FoldersTable.parent.isNull() else (FoldersTable.parent eq parent)
        }.firstOrNull()?.let { return@transaction DavTarget.Collection(it[FoldersTable.id]) }
        FilesTable.selectAll().where {
            (FilesTable.user eq userId) and (FilesTable.name eq last) and (FilesTable.deletedAt eq 0L) and
                if (parent == null) FilesTable.folder.isNull() else (FilesTable.folder eq parent)
        }.firstOrNull()?.let { return@transaction DavTarget.Blob(it[FilesTable.id]) }
        DavTarget.Missing
    }
}
```

```kotlin
    private fun versions(fileId: String) =
        client.get("/api/files/$fileId/versions") { header(HttpHeaders.Authorization, bearer) }
            .let { json.parseToJsonElement(it.bodyAsText()) }.let { e ->
                (e as JsonObject)["versions"]!!.jsonArray.map { it.toString() }
            }
    private fun trashCount() =
        client.get("/api/trash") { header(HttpHeaders.Authorization, bearer) }
            .let { json.parseToJsonElement(it.bodyAsText()) }
            .let { (it as JsonObject)["files"]!!.jsonArray.size }
```
